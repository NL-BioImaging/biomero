# --------------------- CONCURRENCY HELPERS ---------------------------- #

# Messages (lowercase) of database errors that are concurrency conflicts, and
# so worth retrying.
RETRYABLE_CONFLICT_MESSAGES = (
    "unique constraint",
    "duplicate key",
    "concurrent update",
    "transaction is aborted",
    "deadlock detected",
)

# Messages (lowercase) of permanent data validation errors: never retried.
NON_RETRYABLE_ERROR_MESSAGES = (
    "not null constraint",
    "null value in column",
    "cannot be null",
)


def conflict_exception_types() -> tuple:
    """Return the exception types that may be a database conflict."""
    exception_types = [IntegrityError, OperationalError]
    if EventSourcingIntegrityError:
        exception_types.append(EventSourcingIntegrityError)
    if EventSourcingOperationalError:
        exception_types.append(EventSourcingOperationalError)
    return tuple(exception_types)


def is_eventsourcing_error(error: Exception) -> bool:
    """Whether an error comes from the event sourcing persistence layer.

    Those are always concurrency conflicts (newer versions may have empty
    messages) and never data validation errors.
    """
    eventsourcing_types = tuple(
        error_type for error_type in
        (EventSourcingIntegrityError, EventSourcingOperationalError)
        if error_type)
    return isinstance(error, eventsourcing_types)


def is_retryable_conflict(error: Exception) -> bool:
    """Whether a database error is a concurrency conflict worth retrying.

    Args:
        error (Exception): The exception raised by the database operation.

    Returns:
        False for permanent data validation errors, True for an event
        sourcing error or a message that names a known conflict.
    """
    message = str(error).lower()
    # For EventSourcing exceptions, also check the underlying cause
    if getattr(error, "__cause__", None):
        message += " " + str(error.__cause__).lower()
    if any(text in message for text in NON_RETRYABLE_ERROR_MESSAGES):
        logger.debug(f"Non-retryable data validation error: {error}")
        return False
    return (is_eventsourcing_error(error)
            or any(text in message for text in RETRYABLE_CONFLICT_MESSAGES))


def backoff_delay(attempt: int, base_delay: float, max_delay: float) -> float:
    """Return the exponential backoff, with jitter, before the next attempt."""
    return min(base_delay * (2 ** attempt) + random.uniform(0, 0.1), max_delay)


def retry_on_database_conflict(
        max_retries=DATABASE_CONFLICT_DEFAULT_RETRIES,
        base_delay=0.1, max_delay=5.0):
    """Decorator to retry database operations on concurrency conflicts.

    Args:
        max_retries: Maximum number of retry attempts
        base_delay: Base delay between retries in seconds
        max_delay: Maximum delay between retries in seconds
    """
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except conflict_exception_types() as error:
                    last_exception = error
                    if (attempt == max_retries
                            or not is_retryable_conflict(error)):
                        break
                    delay = backoff_delay(attempt, base_delay, max_delay)
                    logger.warning(
                        f"Database conflict on attempt {attempt + 1}/{max_retries + 1}: {error}. "
                        f"Retrying in {delay:.2f}s..."
                    )
                    time.sleep(delay)
                    try:
                        EngineManager.rollback()
                    except Exception as rollback_error:
                        logger.warning(f"Rollback before retry failed: "
                                       f"{rollback_error}")

            # All retries exhausted, raise the last exception
            logger.error(f"Database operation failed after {attempt + 1} attempts (max {max_retries + 1}): {last_exception}")
            raise last_exception

        return wrapper
    return decorator


@contextmanager
def database_transaction(isolation_level=None):
    """Context manager for database transactions with proper error handling.
    
    Args:
        isolation_level: SQLAlchemy isolation level (e.g., 'READ_COMMITTED', 'SERIALIZABLE')
    """
    # Don't manage session lifecycle here - let the scoped session handle it
    # Just provide transaction boundaries and error handling
    try:
        # Set isolation level if specified
        if isolation_level:
            session = EngineManager.get_session()
            session.connection(execution_options={'isolation_level': isolation_level})
            
        yield
        
        # Commit if we reach this point
        EngineManager.commit()
        
    except Exception as e:
        try:
            EngineManager.rollback()
            logger.debug(f"Transaction rolled back due to: {e}")
        except Exception as rollback_error:
            logger.error(f"Error during rollback: {rollback_error}")
        raise

class EngineHolder:

    (Fabric connections are equal when their host is), so each owns a token.
    """
    __slots__ = ("__weakref__",)



[docs]class EngineManager:
    """
    Manages the SQLAlchemy engine and session lifecycle.

    Class Attributes:
        _engine: The SQLAlchemy engine used to connect to the database.
        _scoped_session_topic: The topic of the scoped session.
        _session: The scoped session used for database operations.
        _holders: The live ``EngineHolder`` tokens that acquired the engine.
            Weakly referenced, so a holder that is garbage collected without
            releasing does not keep the engine alive.
    """
    _engine = None
    _scoped_session_topic = None
    _session = None
    _holders = weakref.WeakSet()
    _holders_lock = threading.Lock()


[docs]    @classmethod
    def acquire(cls, holder: EngineHolder, sqlalchemy_url: str = None) -> str:
        """Create the scoped session for ``holder`` and register it as a user.

        Creating the engine and registering the holder happen under one lock,
        so a holder that exits meanwhile cannot dispose an engine that this
        one is about to use.

        Args:
            holder (EngineHolder): The token of the new user.
            sqlalchemy_url (str, optional): The SQLAlchemy database URL; see
                ``create_scoped_session``.

        Returns:
            str: The topic of the scoped session adapter class.
        """
        with cls._holders_lock:
            topic = cls.create_scoped_session(sqlalchemy_url=sqlalchemy_url)
            cls._holders.add(holder)
            return topic



[docs]    @classmethod
    def release(cls, holder: EngineHolder):
        """Unregister ``holder``; dispose the engine with the last holder.

        While other holders remain (the workflow supervisor runs several
        clients), only this thread's session is removed. Deciding and
        disposing happen under the lock of ``acquire``. Releasing a holder
        twice, or one that never acquired, is harmless.

        Args:
            holder (EngineHolder): The token of the user that is done.
        """
        with cls._holders_lock:
            cls._holders.discard(holder)
            if cls._holders:
                logger.debug("Other holders use the database engine; "
                             "releasing only this thread's session")
                cls.remove_session()
            else:
                cls.close_engine()

        
    @classmethod
    def _create_tables(cls):
        """Create the view tables that do not exist yet, and detect fresh installs.

        Concurrent starts can race here, and the loser's DDL may fail although
        the tables exist afterwards. That is the only failure tolerated: after
        a database error, every table must exist, otherwise the original error
        (wrong URL, bad credentials, missing privileges, failed DDL) is raised.
        """
        try:
            Base.metadata.create_all(cls._engine)
        except DBAPIError as create_error:
            try:
                existing = inspect(cls._engine)
                missing = [name for name in Base.metadata.tables
                           if not existing.has_table(name)]
            except DBAPIError:
                # The check failed too: report the error that started it.
                raise create_error
            if missing:
                raise
            logger.debug("Lost the race to create the tables; they exist now")


[docs]    @classmethod
    def create_scoped_session(cls, sqlalchemy_url: str = None):
        """
        Creates and returns a scoped session for interacting with the
        database.

        If the engine doesn't already exist, it initializes the SQLAlchemy
        engine and sets up the scoped session. The engine and the session are
        set up independently: a long-lived process (such as the workflow
        supervisor) can outlive a session that was removed on the way out of a
        SlurmClient context, and still needs a working session afterwards.

        Args:
            sqlalchemy_url (str, optional): The SQLAlchemy database URL. If
                not provided, the method will retrieve the value from the
                'SQLALCHEMY_URL' environment variable.

        Returns:
            str: The topic of the scoped session adapter class.
        """
        if cls._engine is None:
            # Note, we only allow sqlalchemy eventsourcing module
            if not sqlalchemy_url:
                sqlalchemy_url = os.getenv(slurm_env.SQLALCHEMY_URL)
            cls._engine = create_engine(sqlalchemy_url)
            try:
                cls._create_tables()
            except Exception:
                # Do not keep an engine whose tables are missing: the next
                # call would skip creating them.
                cls._engine.dispose()
                cls._engine = None
                raise

        if cls._session is None:
            # Create a scoped_session object.
            cls._session = scoped_session(
                sessionmaker(
                    autocommit=False,
                    autoflush=True,
                    bind=cls._engine,
                )
            )

            class MyScopedSessionAdapter:
                def __getattribute__(self, item: str) -> None:
                    return getattr(cls._session, item)

            # Produce the topic of the scoped session adapter class.
            cls._scoped_session_topic = get_topic(MyScopedSessionAdapter)

        return cls._scoped_session_topic



[docs]    @classmethod
    def get_session(cls):
        """
        Retrieves the current scoped session, setting one up when needed.

        This is a borrowed, thread-scoped session, not a new owned session.
        Do not use it as a context manager for reads made while an event is
        being processed: Session.__exit__ closes the session and rolls back
        the pending producer event transaction. A block that commits before it
        ends (the views' ``update_view_table``) is safe, because nothing is
        pending when it closes. SlurmClient's outer context releases the
        session when work finishes.

        Returns:
            Session: The SQLAlchemy session for interacting with the database.
        """
        if cls._session is None:
            cls.create_scoped_session()
        return cls._session()

    

[docs]    @classmethod
    def commit(cls):
        """
        Commits the current transaction in the scoped session.
        """
        if cls._session is None:
            logger.debug("No database session to commit")
            return
        try:
            cls._session.commit()
        except Exception:
            # The caller (or retry_on_database_conflict) logs the error.
            cls.rollback()
            raise

    

[docs]    @classmethod
    def rollback(cls):
        """
        Rolls back the current transaction in the scoped session.
        """
        if cls._session is None:
            logger.debug("No database session to roll back")
            return
        try:
            cls._session.rollback()
        except Exception:
            # Force session removal on rollback failure; the error is raised
            # for the caller to log.
            cls._session.remove()
            raise

    

[docs]    @classmethod
    def remove_session(cls):
        """
        Removes the current session from the scoped session registry.

        Safe to call when no session was ever set up, so that it can be used
        for cleanup on any exit path.
        """
        if cls._session is not None:
            cls._session.remove()

[docs]    @classmethod
    def close_engine(cls):
        """
        Closes the database engine and cleans up the session.

    This method disposes of the SQLAlchemy engine, removes the session,
        and resets all associated class attributes to `None`.
        """
        if cls._engine is not None:
            cls._session.remove()
            cls._engine.dispose()
            cls._engine = None
            cls._session = None
            cls._scoped_session_topic = None
            clear_topic_cache()
